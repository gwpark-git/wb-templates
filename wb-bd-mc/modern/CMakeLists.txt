cmake_minimum_required(VERSION 3.18)
project(bdmc_modern VERSION 0.1.0 LANGUAGES CXX)

option(BDMC_BUILD_PYTHON "Build the bdmc._core pybind11 module" ON)
option(BDMC_BUILD_TESTS "Build standalone native unit/physics checks" ON)
option(BDMC_ENABLE_SANITIZERS "Enable address/undefined sanitizers with GCC/Clang" OFF)
option(BDMC_ENABLE_OPENMP "Enable explicit OpenMP bulk execution backend" OFF)
option(BDMC_ENABLE_MPI "Build the distributed independent-run MPI executable" OFF)
option(BDMC_ENABLE_CUDA "Build CUDA float64 bulk kernels (requires toolkit/device for tests)" OFF)
option(BDMC_ENABLE_METAL "Build Apple Metal float32 bulk kernels" OFF)
option(BDMC_ENABLE_ACCELERATE "Build Apple Accelerate float64 vector stepping" OFF)
option(BDMC_APPLE_NATIVE "Optimize this Apple Silicon build for the current CPU" OFF)
option(BDMC_BUILD_NETWORK "Build migrated labelled-network components" ON)
option(BDMC_BUILD_HYDRODYNAMICS "Build bounded free-space HI references (requires GSL)" ON)
option(BDMC_BUILD_LEGACY "Build the original simulator through CMake with C++17" OFF)
option(BDMC_NETWORK_LEGACY_ORACLE "Compare migrated network primitives with actual legacy inline laws" OFF)
option(BDMC_BUILD_HARD_SPHERE_MC "Build the standalone periodic NVT hard-sphere sampler" OFF)
option(BDMC_BUILD_BENCHMARKS "Build the opt-in equilibrium transport benchmark" OFF)
set(BDMC_EXTRA_BINDING_SOURCES "" CACHE STRING "Optional binding TUs defining bind_extra(pybind11::module_&)")
set(BDMC_EXTRA_LINK_LIBRARIES "" CACHE STRING "Optional extra libraries for the Python module")
set(BDMC_EXTRA_INCLUDE_DIRECTORIES "" CACHE STRING "Optional include paths for extra binding sources")

if((BDMC_ENABLE_METAL OR BDMC_ENABLE_ACCELERATE OR BDMC_APPLE_NATIVE) AND NOT APPLE)
    message(FATAL_ERROR "Apple Metal/Accelerate/native options require a macOS toolchain; no fallback is selected")
endif()

if(BDMC_ENABLE_SANITIZERS AND (BDMC_ENABLE_CUDA OR BDMC_ENABLE_METAL))
    message(FATAL_ERROR "Run host ASan/UBSan separately from GPU device builds")
endif()
add_library(bdmc_core STATIC src/core.cpp src/spatial.cpp src/reference.cpp src/execution.cpp src/structure.cpp)
if(BDMC_BUILD_NETWORK)
    target_sources(bdmc_core PRIVATE src/network.cpp)
endif()
set_target_properties(bdmc_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(bdmc_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(bdmc_core PUBLIC cxx_std_17)
if(BDMC_BUILD_HYDRODYNAMICS)
    find_package(GSL REQUIRED)
    set(BDMC_HI_VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/hi_native)
    add_library(bdmc_hydrodynamics STATIC src/hydrodynamics.cpp
        ${BDMC_HI_VENDOR}/lib/hi_bd.cpp ${BDMC_HI_VENDOR}/lib/hi_sd.cpp)
    set_target_properties(bdmc_hydrodynamics PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_features(bdmc_hydrodynamics PUBLIC cxx_std_17)
    target_include_directories(bdmc_hydrodynamics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include
        PRIVATE ${BDMC_HI_VENDOR}/lib)
    # Keep GSL's CBLAS provider off pure execution module link lines, especially
    # when Accelerate supplies the same ordinary cblas_* symbols on macOS.
    target_link_libraries(bdmc_hydrodynamics PUBLIC bdmc_core GSL::gsl GSL::gslcblas)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(bdmc_hydrodynamics PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
    endif()
    install(FILES ${BDMC_HI_VENDOR}/third_party/hi_sd/LICENSE
        ${BDMC_HI_VENDOR}/third_party/hi_sd/NOTICE
        ${BDMC_HI_VENDOR}/third_party/hi_sd/SOURCE.json
        DESTINATION bdmc/third_party/hi_sd)
    install(FILES ${BDMC_HI_VENDOR}/IMPORT_MANIFEST.json
        DESTINATION bdmc/third_party)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(bdmc_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX,OBJCXX>:-Wall;-Wextra;-Wpedantic;-fno-fast-math>")
endif()
if(BDMC_ENABLE_OPENMP OR BDMC_BUILD_LEGACY)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
endif()
if(BDMC_ENABLE_OPENMP)
    target_compile_definitions(bdmc_core PUBLIC BDMC_HAVE_OPENMP=1)
    target_link_libraries(bdmc_core PUBLIC OpenMP::OpenMP_CXX)
endif()
if(BDMC_ENABLE_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    target_sources(bdmc_core PRIVATE src/execution_cuda.cu)
    set_target_properties(bdmc_core PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_definitions(bdmc_core PUBLIC BDMC_HAVE_CUDA=1)
    target_compile_options(bdmc_core PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>")
    target_link_libraries(bdmc_core PUBLIC CUDA::cudart)
endif()
if(BDMC_ENABLE_METAL)
    enable_language(OBJCXX)
    target_sources(bdmc_core PRIVATE src/execution_metal.mm)
    set_target_properties(bdmc_core PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
    set_source_files_properties(src/execution_metal.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_compile_definitions(bdmc_core PUBLIC BDMC_HAVE_METAL=1)
    find_library(BDMC_METAL_FRAMEWORK Metal REQUIRED)
    find_library(BDMC_FOUNDATION_FRAMEWORK Foundation REQUIRED)
    target_link_libraries(bdmc_core PUBLIC ${BDMC_METAL_FRAMEWORK} ${BDMC_FOUNDATION_FRAMEWORK})
endif()
if(BDMC_ENABLE_ACCELERATE)
    target_sources(bdmc_core PRIVATE src/execution_accelerate.cpp)
    target_compile_definitions(bdmc_core PUBLIC BDMC_HAVE_ACCELERATE=1)
    find_library(BDMC_ACCELERATE_FRAMEWORK Accelerate REQUIRED)
    target_link_libraries(bdmc_core PUBLIC ${BDMC_ACCELERATE_FRAMEWORK})
endif()
if(BDMC_APPLE_NATIVE)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
        message(FATAL_ERROR "BDMC_APPLE_NATIVE requires an Apple Silicon arm64 build")
    endif()
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-mcpu=native" BDMC_COMPILER_HAS_APPLE_NATIVE)
    if(NOT BDMC_COMPILER_HAS_APPLE_NATIVE)
        message(FATAL_ERROR "The selected Apple compiler does not support -mcpu=native; choose a portable CPU preset explicitly")
    endif()
    target_compile_options(bdmc_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX,OBJCXX>:-mcpu=native>")
endif()
if(BDMC_ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "BDMC_ENABLE_SANITIZERS requires GCC or Clang")
    endif()
    target_compile_options(bdmc_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bdmc_core PUBLIC -fsanitize=address,undefined)
endif()

if(BDMC_BUILD_PYTHON)
    find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
    # Use an installed pybind11; never fetch dependencies during configuration.
    find_package(pybind11 CONFIG QUIET)
    if(NOT pybind11_FOUND)
        execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir
            RESULT_VARIABLE pybind_result OUTPUT_VARIABLE pybind_path
            OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        if(pybind_result EQUAL 0)
            find_package(pybind11 CONFIG REQUIRED PATHS "${pybind_path}" NO_DEFAULT_PATH)
        else()
            message(FATAL_ERROR "Install pybind11 in the selected Python environment")
        endif()
    endif()
    pybind11_add_module(_core src/bindings.cpp)
    target_link_libraries(_core PRIVATE bdmc_core ${BDMC_EXTRA_LINK_LIBRARIES})
    if(BDMC_EXTRA_BINDING_SOURCES)
        target_sources(_core PRIVATE ${BDMC_EXTRA_BINDING_SOURCES})
        target_compile_definitions(_core PRIVATE BDMC_HAS_EXTRA_BINDINGS=1)
        target_include_directories(_core PRIVATE ${BDMC_EXTRA_INCLUDE_DIRECTORIES})
    endif()
    install(TARGETS _core LIBRARY DESTINATION bdmc RUNTIME DESTINATION bdmc)
    pybind11_add_module(_reference src/reference_bindings.cpp)
    target_link_libraries(_reference PRIVATE bdmc_core)
    install(TARGETS _reference LIBRARY DESTINATION bdmc RUNTIME DESTINATION bdmc)
    pybind11_add_module(_analysis src/structure_bindings.cpp)
    target_link_libraries(_analysis PRIVATE bdmc_core)
    install(TARGETS _analysis LIBRARY DESTINATION bdmc RUNTIME DESTINATION bdmc)
    if(BDMC_BUILD_HYDRODYNAMICS)
        pybind11_add_module(_hydrodynamics src/hydrodynamics_bindings.cpp)
        target_link_libraries(_hydrodynamics PRIVATE bdmc_hydrodynamics)
        install(TARGETS _hydrodynamics LIBRARY DESTINATION bdmc RUNTIME DESTINATION bdmc)
    endif()
    if(BDMC_BUILD_NETWORK)
        pybind11_add_module(_network src/network_bindings.cpp)
        target_link_libraries(_network PRIVATE bdmc_core)
        install(TARGETS _network LIBRARY DESTINATION bdmc RUNTIME DESTINATION bdmc)
    endif()
endif()

    target_compile_features(bdmc_legacy_model PUBLIC cxx_std_17)
    target_link_libraries(bdmc_legacy_model PUBLIC GSL::gsl GSL::gslcblas OpenMP::OpenMP_CXX)
    if(BDMC_ENABLE_SANITIZERS)
        target_compile_options(bdmc_legacy_model PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(bdmc_legacy_model PUBLIC -fsanitize=address,undefined)
